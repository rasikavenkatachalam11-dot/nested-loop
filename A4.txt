#include <stdio.h>
int main() {
    int N;
    scanf("%d", &N);
    int marks[N];
    int i = 0;
    while (i < N) {
        scanf("%d", &marks[i]);
        i++;
    }
    int X;
    scanf("%d", &X);
    int count = 0;
    i = 0;
    while (i < N) {
        if (marks[i] == X) {
            count++;
        }
        i++;
    }
    printf("%d", count);
    return 0;
}
