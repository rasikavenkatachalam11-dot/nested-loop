#include <stdio.h>
#include <limits.h>
int secondHighestSalary(int arr[], int n) {
    if (n < 2)
        return -1;
    int max1 = INT_MIN;
    int max2 = INT_MIN;
    for (int i = 0; i < n; i++) {
        if (arr[i] > max1) {
            max2 = max1;
            max1 = arr[i];
        }
        else if (arr[i] < max1 && arr[i] > max2) {
            max2 = arr[i];
        }
    }
    if (max2 == INT_MIN)
        return -1;
    return max2;
}
int main() {
    int n;
    scanf("%d", &n);
    int arr[n];
    for (int i = 0; i < n; i++) {
        scanf("%d", &arr[i]);
    }
    int result = secondHighestSalary(arr, n);
    if (result == -1)
        printf("Second highest salary not found");
    else
        printf("%d", result);
    return 0;
}
