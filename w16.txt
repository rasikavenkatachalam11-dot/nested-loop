#include <stdio.h>
int main() {
    int walletBalance, n;
    int purchase;
    int successfulPurchases = 0;
    int i = 0;
    scanf("%d", &walletBalance);
    scanf("%d", &n);
    while (i < n) {
        scanf("%d", &purchase);
        if (purchase <= walletBalance) {
            walletBalance -= purchase;
            successfulPurchases++;
        } else {
            break;
        }
        i++;
    }
    printf("Successful Purchases: %d\n", successfulPurchases);
    printf("Final Balance: %d\n", walletBalance);
    return 0;
}
