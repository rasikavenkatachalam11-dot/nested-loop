#include <stdio.h>
int main() {
    int n;
    scanf("%d", &n);
    int prev, curr;
    int i = 1;
    scanf("%d", &prev);
    int totalDrops = 0;
    int consecutiveDrops = 0;
    int crashDay = -1;
    while (i < n) {
        scanf("%d", &curr);
        if (curr < prev) {
            totalDrops++;
            consecutiveDrops++;
        } else {
            consecutiveDrops = 0;
        }
        if (consecutiveDrops == 3 && crashDay == -1) {
            crashDay = i + 1;
        }
        prev = curr;
        i++;
    }
    if (crashDay == -1) {
        printf("Crash Day: Not Detected\n");
    } else {
        printf("Crash Day: %d\n", crashDay);
    }
    printf("Total Drops: %d\n", totalDrops);
    return 0;
}
