#include <stdio.h>
int main() {
    int N, K;
    int arr[100], temp[100];
    scanf("%d", &N);
    for (int i = 0; i < N; i++) {
        scanf("%d", &arr[i]);
    }
    scanf("%d", &K);
    for (int i = 0; i < K; i++) {
        temp[i] = arr[i];
    }
    for (int i = K; i < N; i++) {
        arr[i - K] = arr[i];
    }
    for (int i = 0; i < K; i++) {
        arr[N - K + i] = temp[i];
    }
    for (int i = 0; i < N; i++) {
        printf("%d ", arr[i]);
    }
    return 0;
}
