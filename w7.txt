#include <stdio.h>
int main() {
    int capacity, n;
    int change;
    int occupied = 0;
    int criticalCount = 0;
    int criticalLimit;
    scanf("%d", &capacity);
    scanf("%d", &n);
    criticalLimit = (capacity * 90) / 100;
    while (n > 0) {
        scanf("%d", &change);
        occupied = occupied + change;
        if (occupied > criticalLimit) {
            criticalCount = criticalCount + 1;
        }
        n = n - 1;
    }
    printf("Final Occupied Beds: %d\n", occupied);
    printf("Critical Hours: %d\n", criticalCount);
    return 0;
}
