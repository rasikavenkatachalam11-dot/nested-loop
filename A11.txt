#include <stdio.h>
int main() {
    int N;
    int arr[100];
    scanf("%d", &N);
    for (int i = 0; i < N; i++) {
        scanf("%d", &arr[i]);
    }
    int totalSum = 0;
    for (int i = 0; i < N; i++) {
        totalSum += arr[i];
    }
    int leftSum = 0;
    for (int i = 0; i < N; i++) {
        if (leftSum == totalSum - leftSum - arr[i]) {
            printf("%d", i);
            return 0;
        }
        leftSum += arr[i];
    }
    printf("-1");
    return 0;
}
