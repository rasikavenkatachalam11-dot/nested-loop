#include <stdio.h>
#include <limits.h>
int main() {
    int N;
    int price;
    int min = INT_MAX;
    int found = 0;
    scanf("%d", &N);
    for (int i = 0; i < N; i++) {
        scanf("%d", &price);
        if (price > 0 && price < min) {
            min = price;
            found = 1;
        }
    }
    if (found) {
        printf("%d", min);
    } else {
        printf("No positive");
    }
    return 0;
}
