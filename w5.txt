#include <stdio.h>
int main() {
    int n;
    int order;
    int i = 0;
    int successful = 0;
    int cancelled = 0;
    scanf("%d", &n);
    while (i < n) {
        scanf("%d", &order);
        if (order == 1) {
            successful = successful + 1;
        } else {
            cancelled = cancelled + 1;
        }
        i = i + 1;
    }
    printf("Successful: %d\n", successful);
    printf("Cancelled: %d\n", cancelled);
    if (cancelled > successful) {
        printf("Status: Risk\n");
    } else {
        printf("Status: Safe\n");
    }
    return 0;
}
