#include <stdio.h>
int main() {
    int atmCash, n;
    int withdrawal;
    int count = 0;
    int i = 0;
    scanf("%d", &atmCash);
    scanf("%d", &n);
    while (i < n) {
        scanf("%d", &withdrawal);
        if (withdrawal <= atmCash) {
            atmCash -= withdrawal;
            count++;
        } else {
            break;
        }
        i++;
    }
    printf("Successful Withdrawals: %d\n", count);
    printf("Remaining Cash: %d\n", atmCash);
    return 0;
}
