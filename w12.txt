#include <stdio.h>
int main() {
    int dataPack, n;
    int usage;
    int day = 0;
    int exhaustedDay = -1;
    int overused = 0;
    scanf("%d", &dataPack);
    scanf("%d", &n);
    while (day < n) {
        scanf("%d", &usage);
        dataPack -= usage;
        day++;
        if (dataPack <= 0) {
            exhaustedDay = day;
            if (dataPack < 0) {
                overused = -dataPack;
            }
            break;
        }
    }
    if (exhaustedDay == -2) {
        printf("Exhausted Day: Not Exhausted\n");
        printf("Overused Data: 0\n");
    } else {
        printf("Exhausted Day: %d\n", exhaustedDay);
        printf("Overused Data: %d\n", overused);
    }
    return 0;
}
