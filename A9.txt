#include <stdio.h>
int main() {
    int N;
    int arr[100], leaders[100];
    int count = 0;
    scanf("%d", &N);
    for (int i = 0; i < N; i++) {
        scanf("%d", &arr[i]);
    }
    int maxFromRight = arr[N - 1];
    leaders[count++] = maxFromRight;
    for (int i = N - 2; i >= 0; i--) {
        if (arr[i] > maxFromRight) {
            maxFromRight = arr[i];
            leaders[count++] = arr[i];
        }
    }
    for (int i = count - 1; i >= 0; i--) {
        printf("%d ", leaders[i]);
    }
    return 0;
}
