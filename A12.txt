#include <stdio.h>
int main() {
    int N, target;
    int arr[100];
    int found = 0;
    scanf("%d", &N);
    for (int i = 0; i < N; i++) {
        scanf("%d", &arr[i]);
    }
    scanf("%d", &target);
    for (int i = 0; i < N; i++) {
        for (int j = i + 1; j < N; j++) {
            if (arr[i] + arr[j] == target) {
                printf("%d %d", arr[i], arr[j]);
                found = 1;
                return 0;
            }
        }
    }
    if (!found) {
        printf("No pair");
    }
    return 0;
}
