#include <stdio.h>
int main() {
    int maxWeight, n;
    int weight;
    int total = 0;
    int count = 0;
    int i = 0;
    scanf("%d", &maxWeight);
    scanf("%d", &n);
    while (i < n) {
        scanf("%d", &weight);
        if (total + weight > maxWeight) {
            break;
        }
        total = total + weight;
        count = count + 1;
        i = i + 1;
    }
    printf("Passengers Allowed: %d\n", count);
    if (count < n)
        printf("Overload: Yes\n");
    else
        printf("Overload: No\n");
    return 0;
}
