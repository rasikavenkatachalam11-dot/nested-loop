#include <stdio.h>
int main() {
    int N;
    int marks[100];
    int sum = 0;
    float average;
    int count = 0;
    scanf("%d", &N);
    for (int i = 0; i < N; i++) {
        scanf("%d", &marks[i]);
        sum += marks[i];
    }
    average = (float)sum / N;
    for (int i = 0; i < N; i++) {
        if (marks[i] > average) {
            count++;
        }
    }
    printf("%d", count);
    return 0;
}
