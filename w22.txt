#include <stdio.h>
int main() {
    int n;
    scanf("%d", &n);
    int vibration;
    int i = 0;
    int unsafeCount = 0;
    int consecutiveUnsafe = 0;
    int breakdownAt = -1;
    while (i < n) {
        scanf("%d", &vibration);
        if (vibration > 70) {
            unsafeCount++;
            consecutiveUnsafe++;
        } else {
            consecutiveUnsafe = 0;
        }
        if (consecutiveUnsafe == 3 && breakdownAt == -1) {
            breakdownAt = i + 1;
        }
        i++;
    }
    if (breakdownAt == -1) {
        printf("Breakdown At Reading: Not Occurred\n");
    } else {
        printf("Breakdown At Reading: %d\n", breakdownAt);
    }
    printf("Unsafe Readings: %d\n", unsafeCount);
    return 0;
}
