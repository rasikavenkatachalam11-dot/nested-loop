#include <stdio.h>
int main() {
    int N;
    int arr[100];
    scanf("%d", &N);
    for (int i = 0; i < N; i++) {
        scanf("%d", &arr[i]);
    }
    int maxFromRight = -1;
    for (int i = N - 1; i >= 0; i--) {
        int current = arr[i];
        arr[i] = maxFromRight;
        if (current > maxFromRight) {
            maxFromRight = current;
        }
    }
    for (int i = 0; i < N; i++) {
        printf("%d ", arr[i]);
    }
    return 0;
}
